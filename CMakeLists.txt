cmake_minimum_required(VERSION 3.20)
project(Leniascope VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
add_library(lenia_core src/simulation.cpp src/workspace.cpp)
target_include_directories(lenia_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lenia_core PUBLIC Threads::Threads)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Use an x64 build; the selected FFTW import library is x64")
    endif()
    target_link_libraries(lenia_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/fftw_lib/64bit/libfftw3-3.lib")
    add_compile_definitions(NOMINMAX UNICODE _UNICODE)
    add_library(lenia_gpu src/gpu_solver.cpp src/runner.cpp)
    target_link_libraries(lenia_gpu PUBLIC lenia_core d3d11 dxgi d3dcompiler)
    add_executable(LeniascopeCli src/cli.cpp)
    target_link_libraries(LeniascopeCli PRIVATE lenia_gpu)
    add_executable(Leniascope WIN32 src/application.cpp src/desktop_workspace.cpp src/renderer.cpp cmap.cpp
        third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
        third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
        third_party/imgui/backends/imgui_impl_dx11.cpp
        third_party/imgui/backends/imgui_impl_win32.cpp)
    target_include_directories(Leniascope PRIVATE third_party/imgui)
    target_link_libraries(Leniascope PRIVATE lenia_gpu comdlg32 dwmapi shell32)
    function(copy_fftw target)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/fftw_lib/64bit/libfftw3-3.dll" "$<TARGET_FILE_DIR:${target}>")
    endfunction()
    copy_fftw(Leniascope)
    copy_fftw(LeniascopeCli)
    if(BUILD_TESTING)
        add_executable(RenderTests tests/render_tests.cpp src/renderer.cpp cmap.cpp)
        target_link_libraries(RenderTests PRIVATE lenia_gpu)
        copy_fftw(RenderTests)
    endif()
else()
    find_library(FFTW_LIBRARY NAMES fftw3 REQUIRED)
    target_link_libraries(lenia_core PUBLIC "${FFTW_LIBRARY}")
endif()
if(BUILD_TESTING)
    add_executable(CoreTests tests/core_tests.cpp tests/workspace_tests.cpp)
    target_link_libraries(CoreTests PRIVATE lenia_core)
    if(WIN32)
        target_link_libraries(CoreTests PRIVATE lenia_gpu shell32)
        target_sources(CoreTests PRIVATE src/desktop_workspace.cpp)
        copy_fftw(CoreTests)
    endif()
    add_test(NAME core COMMAND CoreTests --cpu-only)
    # Full GPU checks require a hardware adapter; run CoreTests without --cpu-only locally.
endif()
