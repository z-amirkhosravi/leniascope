cmake_minimum_required(VERSION 3.20)
project(Leniascope VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
option(LENIA_ENABLE_CUDA "Build the optional Windows CUDA/cuFFT backend (CUDA 12.8+)" OFF)
if(LENIA_ENABLE_CUDA AND NOT WIN32)
    message(FATAL_ERROR "The optional CUDA display backend currently requires Windows/D3D11")
endif()
add_library(lenia_core src/simulation.cpp src/workspace.cpp src/numerics.cpp src/gallery.cpp src/behavior.cpp src/morphology.cpp src/batch.cpp src/field_edit.cpp)
target_include_directories(lenia_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lenia_core PUBLIC Threads::Threads)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Use an x64 build; the selected FFTW import library is x64")
    endif()
    target_link_libraries(lenia_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/fftw_lib/64bit/libfftw3-3.lib")
    add_compile_definitions(NOMINMAX UNICODE _UNICODE)
    add_library(lenia_gpu src/gpu_solver.cpp src/runner.cpp)
    target_link_libraries(lenia_gpu PUBLIC lenia_core d3d11 dxgi d3dcompiler)
    if(LENIA_ENABLE_CUDA)
        if(CMAKE_VERSION VERSION_LESS 3.21)
            message(FATAL_ERROR "CUDA architecture selection requires CMake 3.21 or later")
        endif()
        enable_language(CUDA)
        find_package(CUDAToolkit 12.8 REQUIRED)
        add_library(lenia_cuda src/cuda_solver.cu)
        set_target_properties(lenia_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
            CUDA_ARCHITECTURES "75-virtual;120-real")
        target_link_libraries(lenia_cuda PUBLIC lenia_core CUDA::cudart CUDA::cufft d3d11 dxgi)
        target_link_libraries(lenia_gpu PUBLIC lenia_cuda)
        target_compile_definitions(lenia_gpu PUBLIC LENIA_ENABLE_CUDA)
    endif()
    add_executable(LeniascopeCli src/cli.cpp)
    target_link_libraries(LeniascopeCli PRIVATE lenia_gpu)
    add_executable(Leniascope WIN32 src/application.cpp src/desktop_workspace.cpp src/gallery_panel.cpp src/gallery_batch.cpp src/gallery_shape.cpp src/editor_panel.cpp src/renderer.cpp cmap.cpp
        third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
        third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
        third_party/imgui/backends/imgui_impl_dx11.cpp
        third_party/imgui/backends/imgui_impl_win32.cpp)
    target_include_directories(Leniascope PRIVATE third_party/imgui)
    target_link_libraries(Leniascope PRIVATE lenia_gpu comdlg32 dwmapi shell32)
    function(copy_fftw target)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/fftw_lib/64bit/libfftw3-3.dll" "$<TARGET_FILE_DIR:${target}>")
        if(LENIA_ENABLE_CUDA)
            file(GLOB cuda_runtime "${CUDAToolkit_BIN_DIR}/cudart64_*.dll"
                "${CUDAToolkit_BIN_DIR}/cufft64_*.dll" "${CUDAToolkit_BIN_DIR}/nvJitLink*.dll")
            foreach(runtime IN LISTS cuda_runtime)
                add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${runtime}" "$<TARGET_FILE_DIR:${target}>")
            endforeach()
        endif()
    endfunction()
    copy_fftw(Leniascope)
    copy_fftw(LeniascopeCli)
    if(BUILD_TESTING)
        add_executable(RenderTests tests/render_tests.cpp src/renderer.cpp cmap.cpp)
        target_link_libraries(RenderTests PRIVATE lenia_gpu)
        copy_fftw(RenderTests)
        add_executable(PipelineBench tests/pipeline_bench.cpp src/renderer.cpp cmap.cpp)
        target_link_libraries(PipelineBench PRIVATE lenia_gpu)
        copy_fftw(PipelineBench)
        if(LENIA_ENABLE_CUDA)
            add_executable(CudaBench tests/cuda_bench.cpp src/renderer.cpp cmap.cpp)
            target_link_libraries(CudaBench PRIVATE lenia_gpu)
            copy_fftw(CudaBench)
        endif()
    endif()
else()
    find_library(FFTW_LIBRARY NAMES fftw3 REQUIRED)
    target_link_libraries(lenia_core PUBLIC "${FFTW_LIBRARY}")
endif()
if(BUILD_TESTING)
    add_executable(CoreTests tests/core_tests.cpp tests/workspace_tests.cpp tests/numerical_tests.cpp tests/seed_tests.cpp tests/gallery_tests.cpp tests/behavior_tests.cpp tests/morphology_tests.cpp tests/batch_tests.cpp tests/field_edit_tests.cpp)
    target_link_libraries(CoreTests PRIVATE lenia_core)
    if(WIN32)
        target_link_libraries(CoreTests PRIVATE lenia_gpu shell32)
        target_sources(CoreTests PRIVATE src/desktop_workspace.cpp)
        copy_fftw(CoreTests)
    endif()
    add_test(NAME core COMMAND CoreTests --cpu-only)
    # Full GPU checks require a hardware adapter; run CoreTests without --cpu-only locally.
endif()
